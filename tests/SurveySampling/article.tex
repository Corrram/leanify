\section{Notation}

We write $A(p,m,z)$ for the sample size for a proportion $p$ with margin of error $m$ at critical value $z$.
We write $B(s,m,z)$ for the sample size for a mean with standard deviation $s$ and margin of error $m$ at critical value $z$.
We write $C(n,N)$ for the sample size $n$ corrected for a population of $N$.
We write $F(n,N)$ for the finite population correction for a sample of $n$ from a population of $N$.
We write $E(p,n,z)$ for the margin of error for a proportion $p$ with sample size $n$ at critical value $z$.
We write $D(k,r)$ for the design effect of clusters of size $k$ with intraclass correlation $r$.
We write $G(n,d)$ for the effective sample size of $n$ observations with design effect $d$.
We write $f(n,N)$ for the sampling fraction of a sample of $n$ from a population of $N$.

\section{Sample sizes}

\begin{lemma}\label{sizes}
$A(\frac12,\frac{1}{10},2)=100$, $B(10,1,2)=400$, $C(1,10)=1$, $F(1,2)=1$ and $F(2,2)=0$.
\end{lemma}

\begin{lemma}\label{margin}
Let $p$, $m$ and $z$ be positive real numbers with $p<1$. Then $E(p,A(p,m,z),z)=m$.
\end{lemma}

\section{Clusters}

\begin{lemma}\label{clusters}
$D(11,\frac{1}{10})=2$, $G(100,2)=50$ and $f(10,100)=\frac{1}{10}$.
\end{lemma}
