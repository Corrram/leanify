\section{Notation}

We write $I_n$ for the identity matrix of size $n$.
We write $p_A$ for the characteristic polynomial of $A$.
We write $a_{ij}(A)$ for the entry of $A$ in row $i$ and column $j$.
We write $M_n$ for the set of $n\times n$ matrices.

\section{Matrices}

\begin{lemma}\label{determinant}
For every positive integer $n$ and all $A,B\in M_n$, $\det(AB)=\det(A)\det(B)$ and $\det(A^\top)=\det(A)$.
\end{lemma}

\begin{lemma}\label{trace}
Let $n$ be a positive integer, and let $A$ and $B$ be $n\times n$ matrices. Then $\operatorname{tr}(AB)=\operatorname{tr}(BA)$ and $\operatorname{tr}(A+B)=\operatorname{tr}(A)+\operatorname{tr}(B)$.
\end{lemma}

\begin{lemma}\label{inverse}
Let $n$ be a positive integer and let $A$ be an invertible $n\times n$ matrix. Then $AA^{-1}=I_n$ and $\det(A^{-1})=\frac{1}{\det(A)}$.
\end{lemma}

\begin{lemma}\label{scalar}
For every positive integer $n$ and every $A\in M_n$, $\det(2A)=2^n\det(A)$.
\end{lemma}

\begin{lemma}\label{rank}
Let $m$ and $n$ be positive integers and let $A$ be an $m\times n$ matrix. Then $\operatorname{rank}(A)\le m$ and $\operatorname{rank}(A^\top)=\operatorname{rank}(A)$.
\end{lemma}

\begin{lemma}\label{charpoly}
Let $n$ be a positive integer and let $A$ be an $n\times n$ matrix. Then $p_A(0)=(-1)^n\det(A)$.
\end{lemma}

\begin{lemma}\label{symmetric}
Every symmetric matrix $A$ satisfies $A^\top=A$, and every orthogonal matrix $Q$ satisfies $\det(Q)^2=1$.
\end{lemma}

\begin{lemma}\label{nilpotent}
Every nilpotent matrix $A$ satisfies $\det(A)=0$ and $\operatorname{tr}(A)=0$.
\end{lemma}

\begin{lemma}\label{implicit-size}
Let $A$ be an $n\times n$ matrix. Then $\det(A^T)=\det A$.
\end{lemma}

\begin{lemma}\label{real-symmetric}
Let $A$ be a real symmetric $n\times n$ matrix. Then $A^T=A$.
\end{lemma}

\begin{lemma}\label{rank-operator}
Let $m$ and $n$ be positive integers and let $A$ be a real $m\times n$ matrix. Then $\operatorname{rank} A\le m$.
\end{lemma}

\begin{lemma}\label{plural-invertible}
Let $A$ and $B$ be invertible $n\times n$ matrices. Then $(AB)^{-1}=B^{-1}A^{-1}$.
\end{lemma}

\begin{lemma}\label{diagonalizable}
Every real symmetric matrix is orthogonally diagonalizable. Every diagonalizable matrix is square. Let $A$ and $B$ be square matrices such that $B$ is similar to $A$. Then $\det A=\det B$.
\end{lemma}

\begin{lemma}\label{similar-invariants}
Similar matrices have the same characteristic polynomial. Let $A$ and $B$ be similar matrices. Then $\det A=\det B$. Let $C$ and $D$ be square matrices. If $C$ and $D$ are similar, then $C$ and $D$ have the same trace.
\end{lemma}

\begin{lemma}\label{eigen-rank}
Let $A$ be an $n\times n$ matrix with $n$ distinct eigenvalues. Then $A$ is diagonalizable. Let $B$ be a matrix. Then the row rank of $B$ equals the column rank of $B$.
\end{lemma}

\begin{lemma}\label{nested-notions}
Let $A$ be a matrix. Then the rank of $A$ equals the rank of the transpose of $A$. The rank of a matrix equals the rank of its transpose. Let $B$ be an invertible matrix. Then the inverse of the transpose of $B$ equals the transpose of the inverse of $B$. Then the determinant of the inverse of $B$ is the reciprocal of the determinant of $B$.
\end{lemma}

\begin{lemma}\label{for-all-formula-noun}
For all $n\times n$ matrices $A$ and $B$, $\det(AB)=\det A\det B$.
\end{lemma}

\begin{lemma}\label{eigenvalues}
Every nilpotent matrix has determinant zero. Every nilpotent matrix has trace $0$. Let $A$ be a square matrix and let $\lambda$ be an eigenvalue of $A$. Then $\lambda^2$ is an eigenvalue of $A^2$. Let $B$ be an invertible matrix. Then $0$ is not an eigenvalue of $B$. Let $C$ be an $n\times n$ matrix with $C^2=C$. Then every eigenvalue of $C$ is $0$ or $1$. Let $v$ be an eigenvector of $C$. Then $v$ is an eigenvector of $C^2$.
\end{lemma}

\begin{lemma}\label{rank-nullity}
Let $A$ be an $m\times n$ matrix. Then the rank of $A$ plus the nullity of $A$ equals $n$. Let $B$ be an invertible $k\times k$ matrix. Then the nullity of $B$ is $0$.
\end{lemma}
