\section{Notation}

We write $\binom{[n]}{k}$ for the number of $k$-element subsets of a set of $n$ elements.
We write $\mathrm{Surj}(k,n)$ for the number of surjective maps from a set of $k$ elements to a set of $n$ elements.
We write $L(m,n)$ for the number of lattice paths from $(0,0)$ to $(m,n)$.
We write $S(n)$ for the number of subsets of a set of $n$ elements.
We write $c(n,k)$ for the number of compositions of $n$ into $k$ positive parts.

\section{Counts}

\begin{lemma}\label{subsets}
$S(3)=8$ and $\binom{[5]}{2}=10$.
\end{lemma}

\begin{lemma}\label{surjections}
$\mathrm{Surj}(3,2)=6$. For every nonnegative integer $n$, $\mathrm{Surj}(n,n)=n!$.
\end{lemma}

\begin{lemma}\label{paths}
$L(2,2)=6$. For all $m,n\in\mathbb{N}$, $L(m,n)=L(n,m)$.
\end{lemma}

\begin{lemma}\label{compositions}
Let $n$ be a positive integer. Then $c(n,1)=1$ and $c(n,n)=1$.
\end{lemma}

\begin{lemma}\label{arrangements}
The number of $2$-permutations of $4$ elements is $12$, and the number of $3$-permutations of $2$ elements is $0$. For every nonnegative integer $n$, the number of permutations of $n$ elements is $n!$.
\end{lemma}

\begin{lemma}\label{maps}
For all $k,n\in\mathbb{N}$, the number of maps from a set of $k$ elements to a set of $n$ elements is $n^k$. The number of injective maps from a set of $3$ elements to a set of $2$ elements is $0$.
\end{lemma}

\begin{lemma}\label{multisets}
The number of $2$-element multisets of $3$ elements is $6$. The number of compositions of $4$ into $2$ nonnegative parts is $5$. The multinomial coefficient of $4$ over $1$, $2$ and $1$ is $12$.
\end{lemma}
