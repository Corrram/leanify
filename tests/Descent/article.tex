\section{Line search}

For every real number $x$, let $f(x)\coloneqq x^2$.

\begin{lemma}\label{direction}
$-1$ is a descent direction for $f$ at $1$.
\end{lemma}

\begin{lemma}\label{armijo}
$\frac12$ is an Armijo step for $f$ at $1$ in direction $-2$ with constant $\frac12$.
\end{lemma}

\begin{lemma}\label{wolfe}
$\frac12$ is a Wolfe step for $f$ at $1$ in direction $-2$ with constants $\frac12$ and $\frac12$.
\end{lemma}

\section{Descent sequences}

\begin{lemma}\label{gradient}
Let $a:\mathbb{R}\to\mathbb{R}$ be a function. If $a$ is a gradient descent sequence for $f$ with step size $\frac14$ and $a(0)=4$, then $a(1)=2$ and $a(2)=1$.
\end{lemma}

\begin{lemma}\label{heavyball}
Let $a:\mathbb{R}\to\mathbb{R}$ be a function. If $a$ is a heavy-ball sequence for $f$ with step size $\frac14$ and momentum $\frac12$ and $a(0)=a(1)=4$, then $a(2)=2$.
\end{lemma}

\begin{lemma}\label{proximal}
Let $a:\mathbb{R}\to\mathbb{R}$ be a function. If $a$ is a proximal point sequence for $f$ with parameter $\frac12$ and $a(0)=2$, then $a(1)=1$.
\end{lemma}
