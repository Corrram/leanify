\section{A condition on some of the variables}

\begin{lemma}\label{unused}
Let $a$, $b$ and $c$ be real numbers with $a\ge0$. Then $a+b\ge b$.
\end{lemma}

\begin{lemma}\label{forall}
For all real numbers $a,b,c$ with $a\ge0$, we have $a+b\ge b$.
\end{lemma}

\begin{lemma}\label{conjunction}
Let $a$ and $b$ be integers with $a\ge0$ and $b\ge0$. Then $a+1>0$.
\end{lemma}

\section{Lists of conditions}

\begin{lemma}\label{comma}
Let $a$ and $b$ be real numbers with $a\ge0$, $b\ge0$. Then $a+b\ge0$.
\end{lemma}

\begin{lemma}\label{list}
Let $a$ and $b$ be real numbers with $a\ge0$, $b\ge0$ and $a<1$. Then $ab<b+1$.
\end{lemma}

\begin{lemma}\label{earlier}
Let $c$ be a positive real number. Let $u$ and $v$ be real numbers with $u\ge0$, $v\ge0$ and $|u-v|<c$. Then $u<v+c$.
\end{lemma}

\begin{lemma}\label{again}
Let $x>0$ and $x<1$. Then $x^2<x$.
\end{lemma}

\begin{lemma}\label{interval}
Let $a$ and $b$ be real numbers with $a\ge0$, $b\in[a,1]$. Then $0\le b\le1$.
\end{lemma}
