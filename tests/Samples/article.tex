\section{Notation}

We write $\bar x_n$ for the sample mean of the first $n$ terms of $x$.
We write $s_n^2(x)$ for the sample variance of the first $n$ terms of $x$.
We write $\hat F_n$ for the empirical distribution function of the first $n$ terms of $x$.
We write $X_{k,n}(x)$ for the $k$th order statistic of the first $n$ terms of $x$.
We write $\operatorname{med}_n(x)$ for the sample median of the first $n$ terms of $x$.
We write $r_n(x,y)$ for the sample correlation coefficient of the first $n$ terms of $x$ and $y$.
We write $b_n(y,x)$ for the slope of the least-squares line of $y$ on $x$ for the first $n$ terms.
For every real number $i$, let $u(i)\coloneqq i$.

\section{Statistics}

\begin{lemma}\label{mean}
$\bar u_3=2$ and $s_3^2(u)=1$.
\end{lemma}

\begin{lemma}\label{variance}
Let $x:\mathbb{R}\to\mathbb{R}$ be a function and let $n\ge2$ be an integer. Then $s_n^2(x)\ge0$.
\end{lemma}

\begin{lemma}\label{order}
$X_{1,3}(u)=1$, $X_{3,3}(u)=3$ and $\operatorname{med}_3(u)=2$.
\end{lemma}

\begin{lemma}\label{correlation}
Let $x,y:\mathbb{R}\to\mathbb{R}$ be functions. If $x(1)\ne x(2)$ and $y(1)\ne y(2)$, then $|r_2(x,y)|\le1$.
\end{lemma}

\begin{lemma}\label{slope}
$b_3(u,u)=1$.
\end{lemma}
