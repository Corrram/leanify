\section{Sums and products over the elements of a finite set}

\begin{lemma}\label{triangle}
Let $A$ be a finite set of real numbers. Then $\sum_{x\in A}x\le\sum_{x\in A}|x|$.
\end{lemma}

\begin{lemma}\label{weights}
Let $V$ be a finite set of real numbers and let $w\colon V\to\mathbb{N}$ be a function. Then for every $U\subseteq V$ we have $\sum_{u\in U}w(u)\ge0$.
\end{lemma}

\begin{lemma}\label{six}
Let $S\coloneqq\{1,2,3\}$. Then $\sum_{x\in S}x=6$ and $\prod_{x\in S}x=6$.
\end{lemma}

\begin{lemma}\label{squares}
Let $n$ be a positive integer. Then $\sum_{k\in\{1,2\}}k^n\ge\prod_{k\in\{1,2\}}k^n$.
\end{lemma}

\begin{lemma}\label{additive}
Let $A$ be a finite set of real numbers and let $f,g\colon A\to\mathbb{R}$ be functions. Then $\sum_{x\in A}(f(x)+g(x))=\sum_{x\in A}f(x)+\sum_{x\in A}g(x)$.
\end{lemma}

\begin{lemma}\label{layout}
Let $A$ be a finite set of real numbers. Then $\sum_{x\in\mathrlap{A}}\ x^2\ge0$.
\end{lemma}

\section{Pairs in a set of pairs}

\begin{lemma}\label{product}
Let $A$ and $B$ be sets of real numbers. Then $(a,b)\in A\times B$ for all $a\in A$ and $b\in B$.
\end{lemma}

\begin{lemma}\label{components}
Let $A$ and $B$ be sets of real numbers. Then $a\in A$ for all $(a,b)\in A\times B$.
\end{lemma}

\begin{lemma}\label{relation}
Let $K$ be a set of pairs of integers. If $(1,2)\in K$, then there is $(x,y)\in K$ such that $x<y$.
\end{lemma}

\begin{lemma}\label{diagonal}
Let $K$ be a set of pairs of real numbers. If $(x,x)\notin K$ for every $x\in\mathbb{R}$, then $x\neq y$ for all $(x,y)\in K$.
\end{lemma}

\begin{lemma}\label{orientations}
Let $V$ be a set of real numbers and let $T$ be a set of pairs of subsets of $V$. Then $A\cup B\subseteq V$ for all $(A,B)\in T$.
\end{lemma}

\begin{lemma}\label{chosen}
Let $V$ be a set of real numbers and let $T$ be a set of pairs of subsets of $V$. Let $(A,B)\in T$. Then $A\cap B\subseteq V$.
\end{lemma}

\begin{lemma}\label{crossing}
Let $V$ be a set of real numbers and let $T$ be a set of pairs of subsets of $V$. Suppose that $A\cup C\neq V$ for all $(A,B)\in T$ and $(C,D)\in T$. Then $A\neq V$ for all $(A,B)\in T$.
\end{lemma}

\section{Notation for pairs}

We write $(A,B)\preceq(C,D)$ if $A\subseteq C$ and $D\subseteq B$.

\begin{lemma}\label{reflexive}
Let $A$ and $B$ be sets of real numbers. Then $(A,B)\preceq(A,B)$.
\end{lemma}

\begin{lemma}\label{transitive}
Let $A$, $B$, $C$, $D$, $E$ and $F$ be sets of real numbers with $(A,B)\preceq(C,D)$ and $(C,D)\preceq(E,F)$. Then $(A,B)\preceq(E,F)$.
\end{lemma}

\begin{lemma}\label{supremum}
Let $A$, $B$, $C$ and $D$ be sets of real numbers. Then $(A,B)\preceq(A\cup C,B\cap D)$.
\end{lemma}

\section{Objects that exist, on a set of the result}

\begin{lemma}\label{positive}
Let $V$ be a set of real numbers. Then there exists a function $w\colon V\to\mathbb{R}$ such that $w(v)>0$ for every $v\in V$.
\end{lemma}

\begin{lemma}\label{decider}
Let $V$ be a finite set of real numbers and let $T$ be a set of pairs of subsets of $V$. Then there exists a function $w\colon V\to\mathbb{R}$ such that $\sum_{u\in A}w(u)\le\sum_{u\in B}w(u)$ for all $(A,B)\in T$.
\end{lemma}

\section{Connectives in parentheses}

\begin{lemma}\label{grouped}
For all real numbers $x$ and $y$, $(x>0\land y>0)\Rightarrow xy>0$.
\end{lemma}

\begin{lemma}\label{either}
For all real numbers $x$ and $y$, $xy=0\Rightarrow(x=0\lor y=0)$.
\end{lemma}
