\section{Notation}

We write $V(G)$ for the vertex set of $G$.
We write $e(G)$ for the number of edges of $G$.
We write $\deg_G(v)$ for the degree of $v$ in $G$.
We write $\chi(G)$ for the chromatic number of $G$.
We write $\omega(G)$ for the clique number of $G$.
We write $\alpha(G)$ for the independence number of $G$.
We write $K_n$ for the complete graph on $n$ vertices.
We write $C_n$ for the cycle on $n$ vertices.

\section{Graphs}

\begin{lemma}\label{handshake}
For every positive integer $n$ and every graph $G$ on $n$ vertices, $\sum_{v=1}^{n}\deg_G(v)=2e(G)$.
\end{lemma}

\begin{lemma}\label{trees}
For every positive integer $n$ and every tree $T$ on $n$ vertices, $e(T)=n-1$.
\end{lemma}

\begin{lemma}\label{clique}
For every graph $G$, $\omega(G)\le\chi(G)$ and $\alpha(G)=\omega(\overline{G})$.
\end{lemma}

\begin{lemma}\label{complete}
For every positive integer $n$, $\chi(K_n)=n$ and $e(K_n)=\frac{n(n-1)}{2}$.
\end{lemma}

\begin{lemma}\label{cycles}
For every integer $n\ge3$, $C_n$ is connected, Hamiltonian and $2$-regular. Moreover, $\chi(C_5)=3$ and $\chi(C_6)=2$.
\end{lemma}

\begin{lemma}\label{bipartite}
Every bipartite graph $G$ with $e(G)\ge1$ satisfies $\chi(G)=2$.
\end{lemma}

\begin{lemma}\label{degree}
Let $n$ be a positive integer and let $G$ be a graph on $n$ vertices. Then $\deg_G(v)\le n-1$ for every $v\in V(G)$.
\end{lemma}

\begin{lemma}\label{regular}
Let $k$ and $n$ be positive integers. For every $k$-regular graph $G$ on $n$ vertices, $2e(G)=kn$.
\end{lemma}

\begin{lemma}\label{tree-edges}
Every tree with $n$ vertices has $n-1$ edges.
\end{lemma}

\begin{lemma}\label{implicit-counts}
Let $G$ be a graph with $n$ vertices and $m$ edges. If $G$ is complete, then $2m=n(n-1)$.
\end{lemma}

\begin{lemma}\label{handshake-odd}
Let $G$ be a graph. Then the number of vertices of odd degree in $G$ is at most the number of vertices of $G$.
\end{lemma}

\begin{lemma}\label{euler}
Let $G$ be a connected graph in which every vertex has even degree. Then $G$ has an Eulerian circuit.
\end{lemma}

\begin{lemma}\label{bipartite-odd-cycle}
Every bipartite graph contains no odd cycle.
\end{lemma}

\begin{lemma}\label{dirac}
Let $G$ be a graph with $n\ge3$ vertices and minimum degree at least $n/2$. Then $G$ has a Hamiltonian cycle.
\end{lemma}

\begin{lemma}\label{bipartite-two}
Every bipartite graph is $2$-colorable.
\end{lemma}

\begin{lemma}\label{leaf-degree}
Let $G$ be a tree and let $v$ be a leaf of $G$. Then $v$ has degree $1$.
\end{lemma}

\begin{lemma}\label{two-leaves}
Every tree with at least two vertices has at least two leaves.
\end{lemma}
