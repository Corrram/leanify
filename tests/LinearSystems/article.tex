\section{Matrices applied to vectors}

\begin{lemma}\label{image}
Let $Q$ be an $n\times\ell$ matrix and let $y$ be a vector in $\mathbb{R}^\ell$. Then $Qy$ is a vector in $\mathbb{R}^n$ and $Q(2y)=2(Qy)$.
\end{lemma}

\begin{lemma}\label{signature}
Let $Q\in\mathbb{R}^{n\times\ell}$ and $b\in\mathbb{R}^\ell$. Then $Q^{\top}Q$ is symmetric and $Qb$ is a vector in $\mathbb{R}^n$.
\end{lemma}

\section{The order of vectors}

\begin{lemma}\label{order}
Let $x$ and $y$ be vectors in $\mathbb{R}^n$ with $x\le y$. Then $x-y\le 0$ and $2x\le 2y$.
\end{lemma}

\begin{lemma}\label{orthant}
Let $x\in\mathbb{R}_{\ge 0}^n$ and $y\in\mathbb{R}_{>0}^n$. Then $x\ge 0$, $y>0$ and $x^{\top}y\ge0$.
\end{lemma}

\begin{lemma}\label{inner}
Let $b$ and $y$ be vectors in $\mathbb{R}^\ell$. Then $b^{\top}y=\langle b,y\rangle$ and $b^Ty=y^Tb$.
\end{lemma}

\section{Alternatives}

\begin{lemma}[Farkas's Lemma]\label{farkas}
Let $Q \in \mathbb{R}^{n \times \ell}$ and $b \in \mathbb{R}^\ell$.
Then exactly one of the following two assertions holds:
\begin{enumerate}
\item There exists $x \in \mathbb{R}_{\ge 0}^n$ with $Q^T x \ge b$.\label{Farkas1}
\item There exists $y \in \mathbb{R}_{\ge 0}^\ell$ with $Qy \le 0$ and $b^T y > 0$.
\end{enumerate}
\end{lemma}

\begin{lemma}\label{trichotomy}
Let $a$ and $b$ be real numbers. Then exactly one of the following two statements holds:
\begin{enumerate}
\item $a\le b$.
\item There exists $c>0$ with $a=b+c$.
\end{enumerate}
\end{lemma}

\begin{lemma}\label{between}
Let $a$ and $b$ be real numbers. Then the following are equivalent:
\begin{enumerate}
\item There exists $x \in \mathbb{R}$ with $a<x$ and $x<b$.
\item $a<b$.
\end{enumerate}
\end{lemma}
