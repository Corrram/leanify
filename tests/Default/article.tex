We write $s(a,b)$ for the nim-sum of $a$ and $b$.
We write $H(n)$ for the number of moves of the Tower of Hanoi with $n$ disks.
We write $\gamma(v,c)$ for the Lorentz factor of the speed $v$ with the speed of light $c$.

\begin{lemma}\label{values}
$s(5,3)=6$, $H(3)=7$ and $\gamma(0,1)=1$.
\end{lemma}

\begin{lemma}\label{unit}
Let $p$ be a real number with $0\le p\le1$. Then $p^2\le p$.
\end{lemma}
