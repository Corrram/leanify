\section{Groups}

\begin{lemma}\label{inverse}
Let $G$ be a group and let $g,h\in G$. Then $(gh)^{-1}=h^{-1}g^{-1}$.
\end{lemma}

\begin{lemma}\label{involution}
Let $G$ be a group and let $g\in G$ with $g^2=1$. Then $g=g^{-1}$.
\end{lemma}

\begin{lemma}\label{normal}
Let $G$ be an abelian group and let $H$ be a subgroup of $G$. Then $H$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{hom}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be a homomorphism. Then $\varphi(1)=1$.
\end{lemma}

\begin{lemma}\label{element}
Let $G$ be a group and let $g$ be an element of $G$. Then $g^{0}=1$.
\end{lemma}

\section{Rings and fields}

\begin{lemma}\label{field}
Let $K$ be a field of characteristic $0$ and let $x\in K$ with $x\ne0$. Then $xx^{-1}=1$.
\end{lemma}

\begin{lemma}\label{ideal}
Let $R$ be a commutative ring and let $I$ be a prime ideal of $R$. Then $I$ is an ideal of $R$.
\end{lemma}

\begin{lemma}\label{distributive}
Let $R$ be a ring and $a,b\in R$. Then $a(b+1)=ab+a$.
\end{lemma}

\section{Vector spaces}

\begin{lemma}\label{scalar}
Let $K$ be a field and let $V$ be a vector space over $K$. Let $v\in V$ and $c\in K$. Then $c(v+v)=cv+cv$.
\end{lemma}

\begin{lemma}\label{linear}
Let $K$ be a field, let $V$ and $W$ be vector spaces over $K$ and let $T\colon V\to W$ be a linear map. Then $T(0)=0$.
\end{lemma}

\begin{lemma}\label{basis}
Let $V$ be a finite-dimensional vector space over a field $K$ and let $B$ be a basis of $V$. Then $B$ is linearly independent.
\end{lemma}

\begin{lemma}\label{kernel}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be an injective homomorphism. Then $\ker\varphi=\{1\}$ and $\ker\varphi$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{kernel-normal}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be a homomorphism. Then the kernel of $\varphi$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{order}
Let $G$ be a group of order $6$. Then the order of $G$ is $6$.
\end{lemma}

\begin{lemma}\label{eigenvalue}
Let $V$ be a vector space over a field $K$, let $T\colon V\to V$ be a linear map and let $c$ be an eigenvalue of $T$. Then $c\in K$.
\end{lemma}

\begin{lemma}\label{contraction}
Let $X$ be a complete metric space and let $f\colon X\to X$ be a contraction. Then there is $x\in X$ such that $f(x)=x$.
\end{lemma}

\begin{lemma}\label{isomorphism}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be an isomorphism. Then $\varphi$ is bijective.
\end{lemma}

\begin{lemma}\label{automorphism}
Let $G$ be a group. Let $\varphi\colon G\to G$ be an automorphism. Then $\varphi(1)=1$.
\end{lemma}

\section{Normed spaces}

\begin{lemma}\label{triangle}
Let $X$ be a normed space and let $x,y\in X$. Then $\|x+y\|\le\|x\|+\|y\|$.
\end{lemma}

\begin{lemma}\label{pythagoras}
Let $H$ be a Hilbert space and let $x,y\in H$ with $\langle x,y\rangle=0$. Then $\|x+y\|^2=\|x\|^2+\|y\|^2$.
\end{lemma}

\begin{lemma}\label{homogeneous}
Let $X$ be a Banach space, let $x\in X$ and let $c$ be a real number. Then $\|cx\|=|c|\|x\|$.
\end{lemma}

\section{Spaces and orders}

\begin{lemma}\label{bounded}
Let $X$ be a compact space and let $f\colon X\to\mathbb{R}$ be a continuous function. Then there is $M\in\mathbb{R}$ such that $f(x)\le M$ for all $x\in X$.
\end{lemma}

\begin{lemma}\label{finite-extension}
Let $K$ be a field and let $L$ be a finite extension of $K$. Then $L$ is algebraic over $K$.
\end{lemma}

\begin{lemma}\label{degree}
Let $L$ be a finitely generated field extension of a field $K$. Then the degree of $L$ over $K$ is at least $1$.
\end{lemma}

\begin{lemma}\label{monoid}
Let $M$ be a monoid and let $a,b,c\in M$. Then $(ab)c=a(bc)$.
\end{lemma}

\begin{lemma}\label{order-two}
Let $G$ be a group and let $g\in G$ be an element of order $2$. Then $g^2=1$.
\end{lemma}

\begin{lemma}\label{algebraic}
Let $K$ be a field, let $L$ be a field extension of $K$ and let $a\in L$ be algebraic over $K$. Then $a\in L$.
\end{lemma}

\begin{lemma}\label{subgroup}
Let $G$ be a group. Let $H$ be a subgroup of $G$. Then $H$ is a group.
\end{lemma}

\begin{lemma}\label{compact}
Let $X$ be a compact Hausdorff space and let $F$ be a closed subset of $X$. Then $F$ is compact.
\end{lemma}

\begin{lemma}\label{order}
Let $P$ be a partially ordered set and let $a,b\in P$ with $a\le b$ and $b\le a$. Then $a=b$.
\end{lemma}

\section{Verbs, complements and real vector spaces}

\begin{lemma}\label{banach}
Let $X$ be a complete metric space and let $f\colon X\to X$ be a contraction. Then $f$ has a unique fixed point.
\end{lemma}

\begin{lemma}\label{identity-in-subgroup}
Let $G$ be a group and let $H$ be a subgroup of $G$. Then the identity of $G$ lies in $H$.
\end{lemma}

\begin{lemma}\label{lagrange}
Let $G$ be a finite group and let $H$ be a subgroup of $G$. Then the order of $H$ divides the order of $G$.
\end{lemma}

\begin{lemma}\label{generated}
Let $G$ be a group and let $g\in G$. Then the subgroup generated by $g$ is abelian.
\end{lemma}

\begin{lemma}\label{abelian-normal}
Let $G$ be an abelian group. Then every subgroup of $G$ is normal.
\end{lemma}

\begin{lemma}\label{abelian-normal-named}
Let $G$ be an abelian group and let $H$ be a subgroup of $G$. Then $H$ is normal.
\end{lemma}

\begin{lemma}\label{injective-surjective}
Let $V$ be a finite-dimensional vector space over $\mathbb{R}$ and let $T\colon V\to V$ be a linear map. If $T$ is injective, then $T$ is surjective.
\end{lemma}

\begin{lemma}\label{dimension}
Let $K$ be a field and let $V$ be a vector space over $K$ of dimension $3$. Then $\dim V=3$.
\end{lemma}

\begin{lemma}\label{element-of-order-two}
Let $G$ be a group of order $6$. Then $G$ has an element of order $2$.
\end{lemma}

\begin{lemma}\label{exponent-two}
Let $G$ be a group. If $g^2=1$ for all $g\in G$, then $G$ is abelian.
\end{lemma}

\begin{lemma}\label{compact-image}
Let $X$ and $Y$ be topological spaces, let $f\colon X\to Y$ be a continuous map and let $K$ be a compact subset of $X$. Then $f(K)$ is compact.
\end{lemma}

\begin{lemma}\label{hausdorff-compact}
Let $X$ be a Hausdorff space. Then every compact subset of $X$ is closed.
\end{lemma}

\begin{lemma}\label{triangle}
Let $X$ be a metric space and let $x,y,z\in X$. Then $d(x,z)\le d(x,y)+d(y,z)$.
\end{lemma}

\begin{lemma}\label{freshman}
Let $K$ be a field of characteristic $p>0$ and let $a\in K$. Then $(a+1)^p=a^p+1$.
\end{lemma}

\begin{lemma}\label{positive-characteristic}
Let $K$ be a field of positive characteristic. Then $K$ is not of characteristic zero.
\end{lemma}

\section{Notions of structures}

\begin{definition}\label{boolean}
A group $G$ is called \emph{boolean} if $g^2=1$ for all $g\in G$.
\end{definition}

\begin{lemma}\label{boolean-abelian}
Every boolean group is abelian.
\end{lemma}

\begin{definition}\label{nice}
We say that a group $G$ is \emph{nice} if $G$ is abelian.
\end{definition}

\begin{lemma}\label{char-hypothesis}
Let $K$ be a field with $\operatorname{char}(K)=p>0$. Then $p>0$.
\end{lemma}

\begin{definition}\label{separated}
We say that a metric space $X$ is \emph{separated} if $d(x,y)\ge1$ for all distinct $x,y\in X$.
\end{definition}

\begin{definition}\label{exponent-two-group}
A group $G$ is called \emph{square-trivial} if $G$ is abelian and $g^2=1$ for all $g\in G$.
\end{definition}

\begin{lemma}\label{finite-algebraic}
Let $L/K$ be a finite field extension and let $a\in L$. Then $a$ is algebraic over $K$.
\end{lemma}

\begin{definition}\label{involution}
Let $G$ be a group. An element $g\in G$ is called an \emph{involution} if $g^2=1$ and $g\neq1$.
\end{definition}

\begin{lemma}\label{involution-inverse}
Let $G$ be a group and let $g\in G$ be an involution. Then $g^{-1}=g$.
\end{lemma}

\begin{lemma}\label{perfection}
Let $K$ be a field of positive characteristic $p>0$. Let $k$ be the perfection of $K$. Then $1\in k$.
\end{lemma}

\begin{lemma}\label{index-two}
Let $G$ be a group and let $H$ be a subgroup of $G$ of index $2$. Then $H$ is normal in $G$.
\end{lemma}

\begin{lemma}\label{index-bracket}
Let $G$ be a group and let $H$ be a subgroup of $G$ with $[G:H]=2$. Then $H$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{finite-index}
Let $G$ be a finite group and let $H$ be a subgroup of $G$. Then $H$ is a subgroup of $G$ of finite index.
\end{lemma}

\begin{lemma}\label{prime-order-cyclic}
Every group of prime order is cyclic.
\end{lemma}

\begin{lemma}\label{finite-order}
Let $G$ be a finite group and let $g\in G$. Then $g$ has finite order.
\end{lemma}

\begin{lemma}\label{odd-order}
Let $G$ be a group of odd order and let $g\in G$. Then $g$ is of odd order.
\end{lemma}

\begin{lemma}\label{cyclic-normal}
Let $G$ be a group and let $H$ be a normal subgroup of $G$ that is cyclic. Then $H$ is abelian.
\end{lemma}

\begin{lemma}\label{cancellation}
Let $S$ be a cancellative semigroup and let $a,b,c\in S$ with $ab=ac$. Then $b=c$.
\end{lemma}

\begin{lemma}\label{idempotent-identity}
Let $G$ be a group and let $e\in G$ be idempotent. Then $e=1$.
\end{lemma}

\begin{lemma}\label{commuting-family}
Let $R$ be a commutative ring and let $x_1,\dots,x_n$ be elements of $R$ with $n\ge2$. Then $x_1x_2=x_2x_1$.
\end{lemma}

\begin{lemma}\label{degree-positive}
Let $L/K$ be a finite field extension. Then $[L:K]\ge1$.
\end{lemma}

\begin{lemma}\label{ideal-ellipsis}
Let $R$ be a commutative ring and $I$ an ideal of $R$. Then $I$ is an ideal of $R$.
\end{lemma}

\begin{lemma}\label{cauchy-bounded}
Let $X$ be a metric space and let $(x_n)$ be a Cauchy sequence in $X$. Then $(x_n)$ is bounded.
\end{lemma}

\begin{lemma}\label{whose-order}
Let $n$ be a positive integer and let $G$ be a group whose order is $n$. Then $G$ is finite.
\end{lemma}

\begin{lemma}\label{domain-satisfy}
Let $R$ be an integral domain. If $a,b\in R$ satisfy $ab=0$, then $a=0$ or $b=0$.
\end{lemma}

\begin{lemma}\label{kernel-trivial}
Let $G$ and $H$ be groups and let $f\colon G\to H$ be a homomorphism of groups. Then the kernel of $f$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{non-abelian}
Let $G$ be a non-abelian group. Then $G$ is non-trivial.
\end{lemma}

\begin{lemma}\label{nonabelian}
Every nonabelian group is nontrivial.
\end{lemma}

\begin{lemma}\label{independent-family}
Let $V$ be a vector space over a field $K$ and let $v_1,\ldots,v_n$ be linearly independent vectors in $V$. Then $v_1,\ldots,v_n$ are linearly independent.
\end{lemma}

\begin{lemma}\label{independent-pair}
Let $V$ be a vector space over a field $K$ and let $v,w\in V$ be linearly independent. Then $v\ne w$.
\end{lemma}

\begin{lemma}\label{let-suppose}
Let $V$ be a vector space over a field $K$, let $v_1,\ldots,v_n\in V$, and suppose that $v_1,\ldots,v_n$ are linearly independent. Then $v_1,\ldots,v_n$ are linearly independent.
\end{lemma}

\begin{lemma}\label{dependent-pair}
Let $V$ be a vector space over a field $K$ and let $u,v$ be vectors in $V$. If $u=v$, then $u$ and $v$ are linearly dependent.
\end{lemma}

\begin{lemma}\label{one-field}
Let $V$ and $W$ be vector spaces over a field $K$ and let $T\colon V\to W$ be a linear map. Then $T$ is linear.
\end{lemma}

\begin{lemma}\label{rank-nullity}
Let $V$ be a finite-dimensional vector space over a field $K$ and let $T\colon V\to V$ be linear. Then $\dim\ker T+\dim\operatorname{im}T=\dim V$.
\end{lemma}

\begin{lemma}\label{nonzero-invertible}
Let $K$ be a field. Then every nonzero element of $K$ is invertible.
\end{lemma}

\begin{lemma}\label{group-elements}
Let $G$ be a group with $6$ elements. Then $G$ is finite.
\end{lemma}

\begin{lemma}\label{metric-pair}
Let $(X,d)$ be a metric space and let $x,y,z\in X$. Then $d(x,z)\le d(x,y)+d(y,z)$.
\end{lemma}

\begin{lemma}\label{subset-of-space}
Let $X$ be a topological space and let $A\subseteq X$ be closed. Then $A\subseteq X$.
\end{lemma}

\begin{lemma}\label{convergent-cauchy}
Let $X$ be a metric space. Then every convergent sequence in $X$ is Cauchy.
\end{lemma}

\begin{lemma}\label{implicit-complement}
Let $G$ be a group and let $H\subseteq G$ be a normal subgroup. Then $H\subseteq G$.
\end{lemma}

\begin{lemma}\label{unit-in-ring}
Let $R$ be a ring and let $a\in R$ be a unit. Then $a\in R$.
\end{lemma}

\begin{lemma}\label{field-elements}
Let $F$ be a field with at least four elements. Then $F$ is a field.
\end{lemma}

\begin{lemma}\label{ring-with-one}
Let $R$ be a commutative ring with $1$. Then $R$ is a ring.
\end{lemma}

\begin{lemma}\label{compact-subsequence}
Let $X$ be a compact metric space and let $(x_n)$ be a sequence in $X$. Then $(x_n)$ has a convergent subsequence.
\end{lemma}

\begin{lemma}\label{cauchy-sequence}
Let $X$ be a metric space and let $(x_n)$ be a convergent sequence in $X$. Then $(x_n)$ is a Cauchy sequence.
\end{lemma}

\begin{lemma}\label{family-of-subgroups}
Let $G$ be a group and let $\mathcal{F}$ be a family of subgroups of $G$. Then every $H\in\mathcal{F}$ is a subgroup of $G$.
\end{lemma}

\begin{lemma}\label{mapping}
Let $G$ be a group and let $\varphi\colon G\to G$ be a mapping. Then $\varphi(1)\in G$.
\end{lemma}

\begin{lemma}\label{hyphen-module}
Let $R$ be a ring and let $M$ be a finitely generated $R$-module. Then $M$ is a module over $R$.
\end{lemma}

\begin{lemma}\label{and-adjectives}
Let $X$ be a complete and separable metric space. Then $X$ is complete.
\end{lemma}

\begin{definition}\label{homogeneous}
We call a metric space $X$ \emph{homogeneous} if for all $x,y\in X$ there is an isometry $f\colon X\to X$ with $f(x)=y$.
\end{definition}

\begin{lemma}\label{homogeneous-space}
Let $X$ be a homogeneous metric space and let $x\in X$. Then there is an isometry $f\colon X\to X$ with $f(x)=x$.
\end{lemma}

\begin{definition}\label{forms-of-definitions}
A group $G$ is called \emph{agreeable} when $G$ is abelian. We say that a group $G$ is \emph{good} provided that $G$ is abelian. If a group $G$ is abelian, then we say that $G$ is \emph{fine}. A group $G$ such that $G$ is abelian is called a \emph{calm group}. A \emph{quiet group} is a group $G$ such that $G$ is abelian. A group is said to be \emph{mild} if it is abelian.
\end{definition}

\begin{lemma}\label{forms-of-definitions-used}
Let $G$ be an agreeable group. Then $G$ is abelian. Let $G$ be a good group. Then $G$ is abelian. Let $G$ be a fine group. Then $G$ is abelian. Let $G$ be a calm group. Then $G$ is abelian. Let $G$ be a quiet group. Then $G$ is abelian. Let $G$ be a mild group. Then $G$ is abelian.
\end{lemma}

\begin{definition}\label{notion-of-a-variable}
Let $G$ be a group and let $H$ be a subgroup of $G$. Then $H$ is called \emph{wide} if $[G:H]=2$.
\end{definition}

\begin{lemma}\label{wide-subgroup}
Let $G$ be a group and let $H$ be a wide subgroup of $G$. Then $H$ is normal in $G$.
\end{lemma}

\begin{theorem}\label{forms-of-binders}
Let $G$ be a finite group, $n$ a positive integer and $H$ a subgroup of $G$. Then $H$ is finite.
\end{theorem}

\begin{lemma}\label{as-in}
Let $G$, $n$ and $H$ be as in Theorem~\ref{forms-of-binders}. Then $H$ is a subgroup of $G$.
\end{lemma}

\begin{lemma}\label{adjective-commas}
Let $X$ be a compact, connected metric space. Then $X$ is complete.
\end{lemma}

\begin{lemma}\label{possibly-infinite}
Let $G$ be a (possibly infinite) group and let $x$, $y$ be two elements of $G$. Then $xy\in G$.
\end{lemma}

\begin{lemma}\label{with-absolute}
Let $G$ and $H$ be groups, with $H$ abelian. Then $gh=hg$ for all $g,h\in H$.
\end{lemma}

\begin{lemma}\label{given-let}
Given a group $G$, let $H$ be a subgroup of $G$. Then $H$ is a group. For a group $K$, let $L$ be a normal subgroup of $K$. Then $L$ is a subgroup of $K$.
\end{lemma}

\begin{lemma}\label{fix-several}
Fix $a>0$, $b>0$ and $c\ge 0$. Then $a+b+c>0$.
\end{lemma}

\begin{lemma}\label{assume-let}
Assume that $a>0$ and let $n$ be a positive integer such that $n>a$. Then $n>0$.
\end{lemma}

\begin{lemma}\label{equivalent-statements}
Let $G$ be a group. Then the following statements are equivalent:
\begin{enumerate}
\item[(i)] $G$ is abelian;
\item[(ii)] $gh=hg$ for all $g,h\in G$.
\end{enumerate}
\end{lemma}

\begin{lemma}\label{neither-nor}
Let $a$ and $b$ be real numbers. Neither $a^2$ nor $b^2$ is negative.
\end{lemma}

\begin{lemma}\label{the-noun-subject}
Let $X$ be a metric space. In particular, the metric space $X$ is a topological space.
\end{lemma}

\begin{lemma}\label{denoted-by}
Let $G$ be a group. We denote by $Z$ the center of $G$. Then $Z$ is a subgroup of $G$. The center of $G$ is denoted by $C$. Then $C$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{proper-subsets}
Let $X$ be a topological space and let $A\subsetneq X$. Then $A\subseteq X$ and $A\neq X$. Let $B$ and $C$ be subsets of $X$ with $B\subsetneq C$. Then $B\subseteq C$.
\end{lemma}

\begin{lemma}\label{old-notation}
Let ${\cal G}$ be a group and let $g\in{\cal G}$. Then $g\in\mathcal{G}$. For every integer $n\ge 2$, we have ${n \choose 2}=\frac{n(n-1)}{2}$. Let $x$ be a real number with $x\not= 0$. Then $x^2\stackrel{(a)}{>} 0$ and $\smash{x}^2\mkern-2mu\ge 0$.
\end{lemma}

\begin{lemma}\label{more-spaces}
Let $X$ be a Polish space. Then $X$ is separable. Let $Y$ be a second countable space. Then $Y$ is a topological space. Let $Z$ be a geodesic metric space. Then $Z$ is a metric space. Let $A$ be a locally closed subset of $Y$. Then $A\subseteq Y$.
\end{lemma}

\begin{lemma}\label{more-maps}
Let $X$ and $Y$ be metric spaces and let $f\colon X\to Y$ be an isometric embedding. Then $d(f(x),f(y))=d(x,y)$ for all $x,y\in X$. Let $g\colon X\to Y$ be a quasi-isometric embedding and let $h\colon X\to Y$ be a quasi-isometry. Then $h$ is a quasi-isometric embedding. Let $u\colon X\to Y$ be an open map and let $v\colon X\to Y$ be a one-to-one map. Then $v$ is injective.
\end{lemma}

\begin{lemma}\label{more-algebra}
Let $R$ be a unital ring. Then $R$ is a ring. Let $G$ be a given group and let $g$ be an arbitrary element of $G$. Then $gg^{-1}=1$. Let $V$ be a vector space over a field $K$ and let $W$ be a $3$-dimensional subspace of $V$. Then $W$ is a subspace of $V$.
\end{lemma}

\begin{lemma}\label{set-of}
Let $G$ be a group. Let $\mathcal{S}$ denote the set of subgroups of $G$. Then $G\in\mathcal{S}$. Let $X$ be a topological space. We denote by $\mathcal{O}$ the set of all open subsets of $X$. Then $X\in\mathcal{O}$.
\end{lemma}

\begin{lemma}\label{chained-binders}
Let $G$ be a group. For every $1\neq g\in G$, we have $gg^{-1}=1$. Let $0<x\in\mathbb{R}$. Then $x^2>0$. We denote the center of $G$ by $Z$. Then $Z$ is a subgroup of $G$.
\end{lemma}

\section{Classical statements}

% An indefinite inside `every`, `let` or `the` is universal: its object is one more variable.

\begin{lemma}\label{nested-indefinites}
Every subgroup of a cyclic group is cyclic.
\end{lemma}

\begin{lemma}\label{nested-subsets}
Every compact subset of a Hausdorff space is closed.
\end{lemma}

\begin{lemma}\label{nested-let}
Let $H$ be a subgroup of a cyclic group. Then $H$ is cyclic.
\end{lemma}

\begin{lemma}\label{nested-signature}
Let $f\colon X\to Y$ be a continuous bijection from a compact space to a Hausdorff space. Then $f$ is a homeomorphism.
\end{lemma}

\begin{lemma}\label{nested-operator}
Let $T\colon V\to V$ be a linear operator on a finite-dimensional vector space. Then $T$ is injective if and only if $T$ is surjective.
\end{lemma}

\begin{lemma}\label{signature-kinds}
Let $\varphi\colon G\to H$ be a group homomorphism. Then $\ker\varphi$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{nested-maps}
Every continuous map from a compact metric space to a metric space is uniformly continuous.
\end{lemma}

\begin{lemma}\label{the-of-a}
The center of a group $G$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{two-objects}
The union of two compact subsets of a topological space is compact. The product of two compact spaces is compact.
\end{lemma}

\begin{lemma}\label{two-normal-subgroups}
Let $G$ be a group. The intersection of two normal subgroups of $G$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{continuous-image}
The continuous image of a connected space is connected.
\end{lemma}

\begin{lemma}\label{generic-statement}
A metric space is compact if and only if it is sequentially compact.
\end{lemma}

\begin{lemma}\label{where-prime}
Let $G$ be a group of order $n^2$, where $n$ is a positive integer. Then $|G|=n^2$.
\end{lemma}

\begin{lemma}\label{implied-complement}
Let $R$ be a commutative ring and let $I$ be a maximal ideal of $R$. Then $I$ is a prime ideal. Every maximal ideal of a commutative ring is prime.
\end{lemma}

\begin{lemma}\label{euclidean}
Every Euclidean domain is a principal ideal domain. Let $R$ be a principal ideal domain and let $a$ be an irreducible element of $R$. Then $a$ is prime.
\end{lemma}

\begin{lemma}\label{vector-space-alone}
Let $V$ be a finite-dimensional vector space. Then $V$ is a vector space. Every path-connected space is connected.
\end{lemma}

\begin{lemma}\label{in-a}
In a principal ideal domain, every irreducible element is prime. The characteristic of a field is either $0$ or positive.
\end{lemma}

\begin{lemma}\label{set-subject}
Let $R$ be a commutative ring. Then the set of nilpotent elements of $R$ is an ideal of $R$. The multiplicative group of a finite field is cyclic.
\end{lemma}

\begin{lemma}\label{rank-nullity}
Let $V$ be a finite-dimensional vector space and let $T\colon V\to W$ be a linear map. Then $\dim V=\dim\ker T+\dim\operatorname{im}T$.
\end{lemma}

\begin{lemma}\label{normed-classical}
Every closed subspace of a Banach space is a Banach space. Let $H$ be a Hilbert space and let $C$ be a nonempty closed convex subset of $H$. Then $C$ is complete.
\end{lemma}

\begin{lemma}\label{operators-between}
Let $T$ be a linear operator between normed spaces. Then $T$ is continuous if and only if $T$ is bounded.
\end{lemma}

\begin{lemma}\label{orthogonal-vectors}
For all vectors $x$ and $y$ in an inner product space, $\|x+y\|^2+\|x-y\|^2=2\|x\|^2+2\|y\|^2$. Let $H$ be an inner product space and let $u$ and $v$ be orthogonal vectors in $H$. Then $\|u+v\|^2=\|u\|^2+\|v\|^2$. Let $(e_n)$ be an orthonormal sequence in a Hilbert space $K$. Then $\|e_1\|=1$.
\end{lemma}

\begin{lemma}\label{index-two}
Every subgroup of index $2$ is normal. Let $n$ be a positive integer and let $G$ be an abelian $n$-group. Then $G$ is a group.
\end{lemma}

\begin{lemma}\label{subspace-sums}
Let $U$ and $W$ be finite-dimensional subspaces of a vector space. Then $\dim(U+W)=\dim U+\dim W-\dim(U\cap W)$. Let $V$ be a vector space and let $X$ and $Y$ be subspaces of $V$. Then $X+Y$ is a subspace of $V$. Let $E$ be a vector space and let $P$ and $Q$ be subspaces of $E$ with $E=P\oplus Q$. Then $\dim E=\dim P+\dim Q$.
\end{lemma}

\begin{lemma}\label{orthonormal-sets}
Let $V$ be an inner product space and let $B$ be an orthonormal subset of $V$. Then $B$ is linearly independent. Let $C$ be an orthogonal subset of $V$ with $0\notin C$. Then $C$ is linearly independent.
\end{lemma}

\begin{lemma}\label{identity-in-subgroup}
Let $G$ be a group and let $H$ be a subgroup of $G$. Then $1\in H$.
\end{lemma}
