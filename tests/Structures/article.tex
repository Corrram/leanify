% Structures with their elements: groups, rings, fields, vector spaces, topological
% spaces and ordered sets, their substructures, and maps bound by their signatures.

\section{Groups}

\begin{lemma}\label{inverse}
Let $G$ be a group and let $g,h\in G$. Then $(gh)^{-1}=h^{-1}g^{-1}$.
\end{lemma}

\begin{lemma}\label{involution}
Let $G$ be a group and let $g\in G$ with $g^2=1$. Then $g=g^{-1}$.
\end{lemma}

\begin{lemma}\label{normal}
Let $G$ be an abelian group and let $H$ be a subgroup of $G$. Then $H$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{hom}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be a homomorphism. Then $\varphi(1)=1$.
\end{lemma}

\begin{lemma}\label{element}
Let $G$ be a group and let $g$ be an element of $G$. Then $g^{0}=1$.
\end{lemma}

\section{Rings and fields}

\begin{lemma}\label{field}
Let $K$ be a field of characteristic $0$ and let $x\in K$ with $x\ne0$. Then $xx^{-1}=1$.
\end{lemma}

\begin{lemma}\label{ideal}
Let $R$ be a commutative ring and let $I$ be a prime ideal of $R$. Then $I$ is an ideal of $R$.
\end{lemma}

\begin{lemma}\label{distributive}
Let $R$ be a ring and $a,b\in R$. Then $a(b+1)=ab+a$.
\end{lemma}

\section{Vector spaces}

\begin{lemma}\label{scalar}
Let $K$ be a field and let $V$ be a vector space over $K$. Let $v\in V$ and $c\in K$. Then $c(v+v)=cv+cv$.
\end{lemma}

\begin{lemma}\label{linear}
Let $K$ be a field, let $V$ and $W$ be vector spaces over $K$ and let $T\colon V\to W$ be a linear map. Then $T(0)=0$.
\end{lemma}

\begin{lemma}\label{basis}
Let $V$ be a finite-dimensional vector space over a field $K$ and let $B$ be a basis of $V$. Then $B$ is linearly independent.
\end{lemma}

\begin{lemma}\label{kernel}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be an injective homomorphism. Then $\ker\varphi=\{1\}$ and $\ker\varphi$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{kernel-normal}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be a homomorphism. Then the kernel of $\varphi$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{order}
Let $G$ be a group of order $6$. Then the order of $G$ is $6$.
\end{lemma}

\begin{lemma}\label{eigenvalue}
Let $V$ be a vector space over a field $K$, let $T\colon V\to V$ be a linear map and let $c$ be an eigenvalue of $T$. Then $c\in K$.
\end{lemma}

\begin{lemma}\label{contraction}
Let $X$ be a complete metric space and let $f\colon X\to X$ be a contraction. Then there is $x\in X$ such that $f(x)=x$.
\end{lemma}

\begin{lemma}\label{isomorphism}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be an isomorphism. Then $\varphi$ is bijective.
\end{lemma}

\begin{lemma}\label{automorphism}
Let $G$ be a group. Let $\varphi\colon G\to G$ be an automorphism. Then $\varphi(1)=1$.
\end{lemma}

\section{Normed spaces}

\begin{lemma}\label{triangle}
Let $X$ be a normed space and let $x,y\in X$. Then $\|x+y\|\le\|x\|+\|y\|$.
\end{lemma}

\begin{lemma}\label{pythagoras}
Let $H$ be a Hilbert space and let $x,y\in H$ with $\langle x,y\rangle=0$. Then $\|x+y\|^2=\|x\|^2+\|y\|^2$.
\end{lemma}

\begin{lemma}\label{homogeneous}
Let $X$ be a Banach space, let $x\in X$ and let $c$ be a real number. Then $\|cx\|=|c|\|x\|$.
\end{lemma}

\section{Spaces and orders}

\begin{lemma}\label{bounded}
Let $X$ be a compact space and let $f\colon X\to\mathbb{R}$ be a continuous function. Then there is $M\in\mathbb{R}$ such that $f(x)\le M$ for all $x\in X$.
\end{lemma}

\begin{lemma}\label{finite-extension}
Let $K$ be a field and let $L$ be a finite extension of $K$. Then $L$ is algebraic over $K$.
\end{lemma}

\begin{lemma}\label{degree}
Let $L$ be a finitely generated field extension of a field $K$. Then the degree of $L$ over $K$ is at least $1$.
\end{lemma}

\begin{lemma}\label{monoid}
Let $M$ be a monoid and let $a,b,c\in M$. Then $(ab)c=a(bc)$.
\end{lemma}

\begin{lemma}\label{order-two}
Let $G$ be a group and let $g\in G$ be an element of order $2$. Then $g^2=1$.
\end{lemma}

\begin{lemma}\label{algebraic}
Let $K$ be a field, let $L$ be a field extension of $K$ and let $a\in L$ be algebraic over $K$. Then $a\in L$.
\end{lemma}

\begin{lemma}\label{subgroup}
Let $G$ be a group. Let $H$ be a subgroup of $G$. Then $H$ is a group.
\end{lemma}

\begin{lemma}\label{compact}
Let $X$ be a compact Hausdorff space and let $F$ be a closed subset of $X$. Then $F$ is compact.
\end{lemma}

\begin{lemma}\label{order}
Let $P$ be a partially ordered set and let $a,b\in P$ with $a\le b$ and $b\le a$. Then $a=b$.
\end{lemma}
