\section{Groups}

\begin{lemma}\label{inverse}
Let $G$ be a group and let $g,h\in G$. Then $(gh)^{-1}=h^{-1}g^{-1}$.
\end{lemma}

\begin{lemma}\label{involution}
Let $G$ be a group and let $g\in G$ with $g^2=1$. Then $g=g^{-1}$.
\end{lemma}

\begin{lemma}\label{normal}
Let $G$ be an abelian group and let $H$ be a subgroup of $G$. Then $H$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{hom}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be a homomorphism. Then $\varphi(1)=1$.
\end{lemma}

\begin{lemma}\label{element}
Let $G$ be a group and let $g$ be an element of $G$. Then $g^{0}=1$.
\end{lemma}

\section{Rings and fields}

\begin{lemma}\label{field}
Let $K$ be a field of characteristic $0$ and let $x\in K$ with $x\ne0$. Then $xx^{-1}=1$.
\end{lemma}

\begin{lemma}\label{ideal}
Let $R$ be a commutative ring and let $I$ be a prime ideal of $R$. Then $I$ is an ideal of $R$.
\end{lemma}

\begin{lemma}\label{distributive}
Let $R$ be a ring and $a,b\in R$. Then $a(b+1)=ab+a$.
\end{lemma}

\section{Vector spaces}

\begin{lemma}\label{scalar}
Let $K$ be a field and let $V$ be a vector space over $K$. Let $v\in V$ and $c\in K$. Then $c(v+v)=cv+cv$.
\end{lemma}

\begin{lemma}\label{linear}
Let $K$ be a field, let $V$ and $W$ be vector spaces over $K$ and let $T\colon V\to W$ be a linear map. Then $T(0)=0$.
\end{lemma}

\begin{lemma}\label{basis}
Let $V$ be a finite-dimensional vector space over a field $K$ and let $B$ be a basis of $V$. Then $B$ is linearly independent.
\end{lemma}

\begin{lemma}\label{kernel}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be an injective homomorphism. Then $\ker\varphi=\{1\}$ and $\ker\varphi$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{kernel-normal}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be a homomorphism. Then the kernel of $\varphi$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{order}
Let $G$ be a group of order $6$. Then the order of $G$ is $6$.
\end{lemma}

\begin{lemma}\label{eigenvalue}
Let $V$ be a vector space over a field $K$, let $T\colon V\to V$ be a linear map and let $c$ be an eigenvalue of $T$. Then $c\in K$.
\end{lemma}

\begin{lemma}\label{contraction}
Let $X$ be a complete metric space and let $f\colon X\to X$ be a contraction. Then there is $x\in X$ such that $f(x)=x$.
\end{lemma}

\begin{lemma}\label{isomorphism}
Let $G$ and $H$ be groups and let $\varphi\colon G\to H$ be an isomorphism. Then $\varphi$ is bijective.
\end{lemma}

\begin{lemma}\label{automorphism}
Let $G$ be a group. Let $\varphi\colon G\to G$ be an automorphism. Then $\varphi(1)=1$.
\end{lemma}

\section{Normed spaces}

\begin{lemma}\label{triangle}
Let $X$ be a normed space and let $x,y\in X$. Then $\|x+y\|\le\|x\|+\|y\|$.
\end{lemma}

\begin{lemma}\label{pythagoras}
Let $H$ be a Hilbert space and let $x,y\in H$ with $\langle x,y\rangle=0$. Then $\|x+y\|^2=\|x\|^2+\|y\|^2$.
\end{lemma}

\begin{lemma}\label{homogeneous}
Let $X$ be a Banach space, let $x\in X$ and let $c$ be a real number. Then $\|cx\|=|c|\|x\|$.
\end{lemma}

\section{Spaces and orders}

\begin{lemma}\label{bounded}
Let $X$ be a compact space and let $f\colon X\to\mathbb{R}$ be a continuous function. Then there is $M\in\mathbb{R}$ such that $f(x)\le M$ for all $x\in X$.
\end{lemma}

\begin{lemma}\label{finite-extension}
Let $K$ be a field and let $L$ be a finite extension of $K$. Then $L$ is algebraic over $K$.
\end{lemma}

\begin{lemma}\label{degree}
Let $L$ be a finitely generated field extension of a field $K$. Then the degree of $L$ over $K$ is at least $1$.
\end{lemma}

\begin{lemma}\label{monoid}
Let $M$ be a monoid and let $a,b,c\in M$. Then $(ab)c=a(bc)$.
\end{lemma}

\begin{lemma}\label{order-two}
Let $G$ be a group and let $g\in G$ be an element of order $2$. Then $g^2=1$.
\end{lemma}

\begin{lemma}\label{algebraic}
Let $K$ be a field, let $L$ be a field extension of $K$ and let $a\in L$ be algebraic over $K$. Then $a\in L$.
\end{lemma}

\begin{lemma}\label{subgroup}
Let $G$ be a group. Let $H$ be a subgroup of $G$. Then $H$ is a group.
\end{lemma}

\begin{lemma}\label{compact}
Let $X$ be a compact Hausdorff space and let $F$ be a closed subset of $X$. Then $F$ is compact.
\end{lemma}

\begin{lemma}\label{order}
Let $P$ be a partially ordered set and let $a,b\in P$ with $a\le b$ and $b\le a$. Then $a=b$.
\end{lemma}

\section{Verbs, complements and real vector spaces}

\begin{lemma}\label{banach}
Let $X$ be a complete metric space and let $f\colon X\to X$ be a contraction. Then $f$ has a unique fixed point.
\end{lemma}

\begin{lemma}\label{identity-in-subgroup}
Let $G$ be a group and let $H$ be a subgroup of $G$. Then the identity of $G$ lies in $H$.
\end{lemma}

\begin{lemma}\label{lagrange}
Let $G$ be a finite group and let $H$ be a subgroup of $G$. Then the order of $H$ divides the order of $G$.
\end{lemma}

\begin{lemma}\label{generated}
Let $G$ be a group and let $g\in G$. Then the subgroup generated by $g$ is abelian.
\end{lemma}

\begin{lemma}\label{abelian-normal}
Let $G$ be an abelian group. Then every subgroup of $G$ is normal.
\end{lemma}

\begin{lemma}\label{abelian-normal-named}
Let $G$ be an abelian group and let $H$ be a subgroup of $G$. Then $H$ is normal.
\end{lemma}

\begin{lemma}\label{injective-surjective}
Let $V$ be a finite-dimensional vector space over $\mathbb{R}$ and let $T\colon V\to V$ be a linear map. If $T$ is injective, then $T$ is surjective.
\end{lemma}

\begin{lemma}\label{dimension}
Let $K$ be a field and let $V$ be a vector space over $K$ of dimension $3$. Then $\dim V=3$.
\end{lemma}

\begin{lemma}\label{element-of-order-two}
Let $G$ be a group of order $6$. Then $G$ has an element of order $2$.
\end{lemma}

\begin{lemma}\label{exponent-two}
Let $G$ be a group. If $g^2=1$ for all $g\in G$, then $G$ is abelian.
\end{lemma}

\begin{lemma}\label{compact-image}
Let $X$ and $Y$ be topological spaces, let $f\colon X\to Y$ be a continuous map and let $K$ be a compact subset of $X$. Then $f(K)$ is compact.
\end{lemma}

\begin{lemma}\label{hausdorff-compact}
Let $X$ be a Hausdorff space. Then every compact subset of $X$ is closed.
\end{lemma}

\begin{lemma}\label{triangle}
Let $X$ be a metric space and let $x,y,z\in X$. Then $d(x,z)\le d(x,y)+d(y,z)$.
\end{lemma}

\begin{lemma}\label{freshman}
Let $K$ be a field of characteristic $p>0$ and let $a\in K$. Then $(a+1)^p=a^p+1$.
\end{lemma}

\begin{lemma}\label{positive-characteristic}
Let $K$ be a field of positive characteristic. Then $K$ is not of characteristic zero.
\end{lemma}

\section{Notions of structures}

\begin{definition}\label{boolean}
A group $G$ is called \emph{boolean} if $g^2=1$ for all $g\in G$.
\end{definition}

\begin{lemma}\label{boolean-abelian}
Every boolean group is abelian.
\end{lemma}

\begin{definition}\label{nice}
We say that a group $G$ is \emph{nice} if $G$ is abelian.
\end{definition}

\begin{lemma}\label{char-hypothesis}
Let $K$ be a field with $\operatorname{char}(K)=p>0$. Then $p>0$.
\end{lemma}

\begin{definition}\label{separated}
We say that a metric space $X$ is \emph{separated} if $d(x,y)\ge1$ for all distinct $x,y\in X$.
\end{definition}

\begin{definition}\label{exponent-two-group}
A group $G$ is called \emph{square-trivial} if $G$ is abelian and $g^2=1$ for all $g\in G$.
\end{definition}

\begin{lemma}\label{finite-algebraic}
Let $L/K$ be a finite field extension and let $a\in L$. Then $a$ is algebraic over $K$.
\end{lemma}

\begin{definition}\label{involution}
Let $G$ be a group. An element $g\in G$ is called an \emph{involution} if $g^2=1$ and $g\neq1$.
\end{definition}

\begin{lemma}\label{involution-inverse}
Let $G$ be a group and let $g\in G$ be an involution. Then $g^{-1}=g$.
\end{lemma}

\begin{lemma}\label{perfection}
Let $K$ be a field of positive characteristic $p>0$. Let $k$ be the perfection of $K$. Then $1\in k$.
\end{lemma}

\begin{lemma}\label{index-two}
Let $G$ be a group and let $H$ be a subgroup of $G$ of index $2$. Then $H$ is normal in $G$.
\end{lemma}

\begin{lemma}\label{index-bracket}
Let $G$ be a group and let $H$ be a subgroup of $G$ with $[G:H]=2$. Then $H$ is a normal subgroup of $G$.
\end{lemma}

\begin{lemma}\label{finite-index}
Let $G$ be a finite group and let $H$ be a subgroup of $G$. Then $H$ is a subgroup of $G$ of finite index.
\end{lemma}

\begin{lemma}\label{prime-order-cyclic}
Every group of prime order is cyclic.
\end{lemma}

\begin{lemma}\label{finite-order}
Let $G$ be a finite group and let $g\in G$. Then $g$ has finite order.
\end{lemma}

\begin{lemma}\label{odd-order}
Let $G$ be a group of odd order and let $g\in G$. Then $g$ is of odd order.
\end{lemma}

\begin{lemma}\label{cyclic-normal}
Let $G$ be a group and let $H$ be a normal subgroup of $G$ that is cyclic. Then $H$ is abelian.
\end{lemma}

\begin{lemma}\label{cancellation}
Let $S$ be a cancellative semigroup and let $a,b,c\in S$ with $ab=ac$. Then $b=c$.
\end{lemma}

\begin{lemma}\label{idempotent-identity}
Let $G$ be a group and let $e\in G$ be idempotent. Then $e=1$.
\end{lemma}

\begin{lemma}\label{commuting-family}
Let $R$ be a commutative ring and let $x_1,\dots,x_n$ be elements of $R$ with $n\ge2$. Then $x_1x_2=x_2x_1$.
\end{lemma}

\begin{lemma}\label{degree-positive}
Let $L/K$ be a finite field extension. Then $[L:K]\ge1$.
\end{lemma}

\begin{lemma}\label{ideal-ellipsis}
Let $R$ be a commutative ring and $I$ an ideal of $R$. Then $I$ is an ideal of $R$.
\end{lemma}

\begin{lemma}\label{cauchy-bounded}
Let $X$ be a metric space and let $(x_n)$ be a Cauchy sequence in $X$. Then $(x_n)$ is bounded.
\end{lemma}

\begin{lemma}\label{whose-order}
Let $n$ be a positive integer and let $G$ be a group whose order is $n$. Then $G$ is finite.
\end{lemma}

\begin{lemma}\label{domain-satisfy}
Let $R$ be an integral domain. If $a,b\in R$ satisfy $ab=0$, then $a=0$ or $b=0$.
\end{lemma}

\begin{lemma}\label{kernel-trivial}
Let $G$ and $H$ be groups and let $f\colon G\to H$ be a homomorphism of groups. Then the kernel of $f$ is a normal subgroup of $G$.
\end{lemma}
