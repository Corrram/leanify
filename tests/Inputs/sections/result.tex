\begin{lemma}\label{lem:positive}
For every $x\in\R$, $r(x)\ge1$.
\end{lemma}

\begin{proof}
Let $x\in\R$.
Then $\sq(x)\ge0$.
Hence $r(x)=\sq(x)+1\ge1$.
\end{proof}

\begin{proposition}\label{prop:minimum}
For every $x\in\R$, $r(0)\le r(x)$.
\end{proposition}

\begin{proof}
The value at the origin is the least one.
By Lemma~\ref{lem:positive}, $r(0)=1\le t(x)$ for every $x\in\R$.
\end{proof}
