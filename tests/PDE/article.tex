A test of fields on $\mathbb R^3\times[0,1)$: vector and scalar fields, function
spaces, compact sets, supports, norms, energies, and the Navier--Stokes equations,
in the words of Theorem~1.1 of the Navier--Stokes blowup paper.

We write $A\subset B$ if $A\subseteq B$.

\begin{lemma}\label{sets}
There exist a compact set $K\subset\mathbb R^3$ and a compact set $L\subset\mathbb R^3$
such that $K\subset L$.
\end{lemma}

\begin{lemma}\label{force}
There is a force $f\in C_c^\infty(\mathbb R^3\times(0,\infty);\mathbb R^3)$.
\end{lemma}

\begin{lemma}\label{fields}
There exist smooth velocity and pressure fields $u,p$ on $\mathbb R^3\times[0,1)$
satisfying
\begin{equation}
  \nabla\cdot u=0,
  \qquad u(\cdot,0)=0,
  \label{eq:div}
\end{equation}
such that $\sup_{0\leq t<1}\lVert u(t)\rVert_{L^2(\mathbb R^3)}<\infty$.
\end{lemma}

\begin{theorem}\label{blowup}
For every $\nu>0$ there exist a force
$f\in C_c^\infty(\mathbb R^3\times(0,\infty);\mathbb R^3)$, a compact set
$K\subset\mathbb R^3$, and smooth velocity and pressure fields
$u,p$ on $\mathbb R^3\times[0,1)$ satisfying
\begin{equation}
  \partial_tu+(u\cdot\nabla)u-\nu\Delta u+\nabla p=f,
  \qquad \nabla\cdot u=0,
  \qquad u(\cdot,0)=0,
  \label{eq:ns}
\end{equation}
such that $\supp u(\cdot,t)\cup\supp p(\cdot,t)\subset K$ for every
$0\leq t<1$,
\begin{equation*}
  \sup_{0\leq t<1}\lVert u(t)\rVert_{L^2(\mathbb R^3)}<\infty,
  \qquad
  \limsup_{t\uparrow1}\lVert u(t)\rVert_{L^\infty(\mathbb R^3)}=\infty,
\end{equation*}
and consequently there are no smooth velocity and pressure fields $u,p$ on
$\mathbb R^3\times[0,\infty)$ satisfying \eqref{eq:ns} such that
$\sup_{t\geq0}\frac12\int_{\mathbb R^3}|u(x,t)|^2\,dx<\infty$.
\end{theorem}

On the torus $\T^3$, a field is a function on $\R^3$ of period $1$ in each spatial
coordinate; norms are taken over a period.

\begin{theorem}\label{periodic}
For every $\nu>0$ there exist a force $f\in C_c^\infty(\T^3\times(0,\infty);\R^3)$, a
compact set $K\subset(-1/2,1/2)^3$, and smooth velocity and pressure fields
$U,P_{\mathrm{per}}$ on $\T^3\times[0,1)$ satisfying
\begin{equation}
  \partial_tU+(U\cdot\nabla)U-\nu\Delta U+\nabla P_{\mathrm{per}}=f,
  \qquad \nabla\cdot U=0,
  \qquad U(\cdot,0)=0,
  \label{eq:per}
\end{equation}
such that $\supp U(\cdot,t)\cup\supp P_{\mathrm{per}}(\cdot,t)\subset K$ for every
$0\le t<1$,
\[
  \limsup_{t\uparrow1}\|U(t)\|_{L^\infty(\T^3)}=\infty,
\]
and consequently there are no smooth velocity and pressure fields
$U,P_{\mathrm{per}}$ on $\T^3\times[0,\infty)$ satisfying \eqref{eq:per}.
\end{theorem}
