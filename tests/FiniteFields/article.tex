\section{Notation}

We write $g(n,q)$ for the order of the general linear group of degree $n$ over the field with $q$ elements.
We write $s(n,q)$ for the order of the special linear group of degree $n$ over the field with $q$ elements.
We write $M(n,q)$ for the number of square matrices of order $n$ over the field with $q$ elements.
We write $G(k,n,q)$ for the number of $k$-dimensional subspaces of a space of dimension $n$ over the field with $q$ elements.
We write $P(n,q)$ for the number of points of the projective space of dimension $n$ over the field with $q$ elements.
We write $E(q)$ for the number of primitive elements of the field with $q$ elements.

\section{Counts}

\begin{lemma}\label{groups}
$g(2,2)=6$, $s(2,3)=24$ and $M(2,2)=16$.
\end{lemma}

\begin{lemma}\label{subspaces}
$G(1,2,2)=3$, $P(2,2)=7$, $E(4)=2$ and $E(8)=6$.
\end{lemma}

\begin{lemma}\label{powers}
$4$ is a prime power and $9$ is a prime power.
\end{lemma}

\begin{lemma}\label{lines}
Let $q$ be a prime power. Then $G(1,2,q)=P(1,q)$.
\end{lemma}

\section{Closed forms}

\begin{lemma}\label{closed}
Let $q$ be a prime power and let $n$ be a positive integer. Then $M(n,q)=q^{n^2}$, $g(n,q)=\prod_{k=0}^{n-1}(q^n-q^k)$ and $g(n,q)=(q-1)s(n,q)$. Moreover, $E(q)$ is positive and $G(n+1,n,q)=0$.
\end{lemma}
