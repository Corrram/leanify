\section{Notation}

We write $\|f\|_\infty$ for the supremum norm of $f$.
We write $\operatorname{Lip}(f)$ for the Lipschitz constant of $f$.
We write $V(f)$ for the total variation of $f$ on $[0,1]$.
We write $Z(f)$ for the set of zeros of $f$.

\section{Real functions}

\begin{lemma}\label{lipschitz}
Let $f$ be a Lipschitz function. Then $f$ is uniformly continuous.
\end{lemma}

\begin{lemma}\label{derivative}
Let $f$ be a differentiable function. If $|f'(x)|\le1$ for every $x\in\mathbb{R}$, then $f$ is $1$-Lipschitz and $\operatorname{Lip}(f)\le1$.
\end{lemma}

\begin{lemma}\label{periodic}
Let $f$ be a continuous periodic function. Then $f$ is bounded and uniformly continuous.
\end{lemma}

\begin{lemma}\label{even}
Let $f$ be an even differentiable function. Then $f'(0)=0$.
\end{lemma}

\begin{lemma}\label{monotone}
Let $f$ be an increasing function. Then $V(f)=f(1)-f(0)$.
\end{lemma}

\begin{lemma}\label{convex}
Let $f$ be a convex function. Then $f$ is continuous.
\end{lemma}

\begin{lemma}\label{bounded}
For every bounded function $f$ and every $x\in\mathbb{R}$, $|f(x)|\le\|f\|_\infty$.
\end{lemma}

\begin{lemma}\label{injective}
Let $f$ be a continuous injective function. Then $f$ is strictly increasing or $f$ is strictly decreasing.
\end{lemma}

\begin{lemma}\label{zeros}
Let $f$ be a strictly increasing function, and let $a\in Z(f)$ and $b\in Z(f)$. Then $a=b$.
\end{lemma}

\begin{lemma}\label{lipschitzconstant}
Every odd function $f$ satisfies $f(0)=0$. Moreover, every odd function $f$ is bounded or surjective.
\end{lemma}

\begin{lemma}\label{parity}
Let $f$ be a function. Then $f$ is constant if and only if $f$ is even and $f$ is monotone.
\end{lemma}

\begin{lemma}\label{function-on-interval}
Let $f$ be a continuous function on $[0,1]$ with $f(0)<0<f(1)$. Then there is $c\in(0,1)$ with $f(c)=0$.
\end{lemma}

\begin{lemma}\label{fixed-point}
Let $f\colon[0,1]\to[0,1]$ be continuous. Then $f$ has a fixed point.
\end{lemma}

\begin{lemma}\label{pointwise-order}
Let $f\colon[0,1]\to\mathbb{R}$ be a continuous function with $f\ge0$. Then $f(1)\ge0$.
\end{lemma}

\begin{lemma}\label{pointwise-bound}
Let $f\colon\mathbb{R}\to\mathbb{R}$ be a function with $f\le1$. Then $f(0)\le1$.
\end{lemma}

\begin{lemma}\label{uniform-on-interval}
Let $f\colon[0,1]\to\mathbb{R}$ be Lipschitz. Then $f$ is uniformly continuous.
\end{lemma}

\begin{lemma}\label{suppose-for-all}
Let $f\colon\mathbb{R}\to\mathbb{R}$ be a function. Suppose that $f(x)\ge0$ for all $x$. Then $f(1)\ge0$.
\end{lemma}

\begin{lemma}\label{mean-value}
Let $a<b$ and let $f\colon[a,b]\to\mathbb{R}$ be continuous on $[a,b]$ and differentiable on $(a,b)$. Then there exists $c\in(a,b)$ with $f'(c)=\frac{f(b)-f(a)}{b-a}$.
\end{lemma}

\begin{lemma}\label{whenever}
Let $\varepsilon>0$. Then there exists $\delta>0$ such that $|x^2-1|<\varepsilon$ whenever $|x-1|<\delta$.
\end{lemma}

\begin{lemma}\label{plural-functions-on}
Let $f$ and $g$ be convex functions on $\mathbb{R}$. Then $f+g$ is convex.
\end{lemma}

\begin{lemma}\label{function-space}
Let $f\in C([0,1])$. Then $f$ is bounded.
\end{lemma}

\begin{lemma}\label{function-space-one}
Let $f\in C^1([0,2])$. Then $f$ is continuous.
\end{lemma}

\begin{lemma}\label{continuous-at}
Let $f\colon\mathbb{R}\to\mathbb{R}$ be continuous at $0$ with $f(0)>0$. Then $f(0)\ge0$.
\end{lemma}

\begin{lemma}\label{the-function}
The function $f(x)=x^3$ is injective, and the function $x\mapsto x^2$ is convex.
\end{lemma}

\begin{lemma}\label{injective-mapping}
Let $f\colon\mathbb{R}\to\mathbb{R}$ be an injective mapping. Then $f$ is injective.
\end{lemma}

\section{Classical statements}

\begin{lemma}\label{riemann}
Let $a<b$ be real numbers. Every continuous function on $[a,b]$ is Riemann integrable. Let $g$ be a monotone function. Then $g$ is Riemann integrable on $[a,b]$.
\end{lemma}

\begin{lemma}\label{rolle}
Let $a<b$ be real numbers and let $f\colon[a,b]\to\mathbb{R}$ be continuous on $[a,b]$ and differentiable on $(a,b)$ with $f(a)=f(b)$. Then there exists $c\in(a,b)$ with $f'(c)=0$.
\end{lemma}

\begin{lemma}\label{extreme-values}
Let $a<b$ be real numbers. Every continuous function $f\colon[a,b]\to\mathbb{R}$ attains its maximum and its minimum. Let $g$ be a continuous function. Then $g$ attains its minimum on $[a,b]$.
\end{lemma}

\begin{lemma}\label{constant-on}
Let $a<b$ be real numbers and let $f\colon[a,b]\to\mathbb{R}$ be differentiable with $f'(x)=0$ for all $x\in[a,b]$. Then $f$ is constant on $[a,b]$.
\end{lemma}

\begin{lemma}\label{onto}
Let $f\colon[0,3]\to[1,2]$ be a surjective function. Then $f(0)\le2$. Let $g\colon[0,3]\to[0,3]$ be bijective. Then $g$ is injective.
\end{lemma}

\begin{lemma}\label{fundamental-theorem}
Let $a<b$ be real numbers, let $f\colon[a,b]\to\mathbb{R}$ be continuous and let $F(x)=\int_a^x f(t)\,dt$. Then $F$ is differentiable on $(a,b)$ and $F'(x)=f(x)$ for all $x\in(a,b)$.
\end{lemma}

\begin{lemma}\label{attains-on}
Let $a<b$. Then every continuous function on $[a,b]$ attains its maximum. A differentiable function is continuous.
\end{lemma}
